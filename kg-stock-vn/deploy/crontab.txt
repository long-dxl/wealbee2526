CRON_TZ=Asia/Ho_Chi_Minh

# 06:00 — Crawl tin tức tài chính (pipeline_runner)

30 15 * * 1-5 cd __APP__/toolcrawldata && __VENV__/bin/python seed_prices_dnse.py     >> /var/log/wealbee/prices.log  2>&1
